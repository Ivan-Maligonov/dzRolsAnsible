nano playbook.yml
- name: dzRols
  hosts: all
  become: yes

  vars:
    drupal_root: /var/www/drupal
    drupal_db: drupal
    drupal_user: drupal
    drupal_pass: drupalpass

  tasks:

  # ---------------- SYSTEM ----------------
  - name: Install required packages
    apt:
      name:
        - nginx
        - postgresql
        - php8.1-fpm
        - php8.1-pgsql
        - php8.1-xml
        - php8.1-mbstring
        - php8.1-gd
        - php8.1-curl
        - unzip
        - wget
      update_cache: yes

  - name: Start services
    service:
      name: "{{ item }}"
      state: started
      enabled: yes
    loop:
      - nginx
      - postgresql
      - php8.1-fpm

  # ---------------- POSTGRES ----------------
  - name: Create database
    become_user: postgres
    postgresql_db:
      name: "{{ drupal_db }}"

  - name: Create user
    become_user: postgres
    postgresql_user:
      name: "{{ drupal_user }}"
      password: "{{ drupal_pass }}"

  - name: Grant privileges
    become_user: postgres
    postgresql_privs:
      db: "{{ drupal_db }}"
      role: "{{ drupal_user }}"
      type: database
      privs: ALL

  # ---------------- DRUPAL ----------------
  - name: Download Drupal
    get_url:
      url: "https://ftp.drupal.org/files/projects/drupal-9.5.11.tar.gz"
      dest: /tmp/drupal.tar.gz

  - name: Extract Drupal
    unarchive:
      src: /tmp/drupal.tar.gz
      dest: /tmp/
      remote_src: yes

  - name: Find extracted folder
    find:
      paths: /tmp
      patterns: "drupal-*"
      file_type: directory
    register: drupal_src

  - name: Clean web root
    file:
      path: "{{ drupal_root }}"
      state: absent

  - name: Create web root
    file:
      path: "{{ drupal_root }}"
      state: directory
      owner: www-data
      group: www-data
      mode: "0755"

  - name: Copy Drupal files
    copy:
      src: "{{ drupal_src.files[0].path }}/"
      dest: "{{ drupal_root }}/"
      remote_src: yes

  - name: Set permissions
    file:
      path: "{{ drupal_root }}"
      state: directory
      recurse: yes
      owner: www-data
      group: www-data

  # ---------------- NGINX ----------------
  - name: Configure nginx vhost
    copy:
      dest: /etc/nginx/sites-available/drupal
      content: |
        server {
            listen 80;
            server_name _;
            root {{ drupal_root }};

            index index.php index.html;

            location / {
                try_files $uri /index.php?$query_string;
            }

            location ~ \.php$ {
                include snippets/fastcgi-php.conf;
                fastcgi_pass unix:/run/php/php8.1-fpm.sock;
            }
        }

  - name: Enable site
    file:
      src: /etc/nginx/sites-available/drupal
      dest: /etc/nginx/sites-enabled/drupal
      state: link

  - name: Remove default nginx site
    file:
      path: /etc/nginx/sites-enabled/default
      state: absent

  - name: Restart nginx
    service:
      name: nginx
      state: restarted

  # ---------------- DRUPAL SETTINGS ----------------
  - name: Create settings.php
    copy:
      dest: "{{ drupal_root }}/sites/default/settings.php"
      content: |
        <?php
        $databases['default']['default'] = [
          'database' => '{{ drupal_db }}',
          'username' => '{{ drupal_user }}',
          'password' => '{{ drupal_pass }}',
          'host' => 'localhost',
          'driver' => 'pgsql',
        ];
